1. Create `pyproject.toml` with these exact sections:

```toml
[project]
name = "helix"
version = "0.1.0"
requires-python = ">=3.12"

[tool.pytest.ini_options]
testpaths = ["tests"]
addopts = "--cov=src --cov-report=term-missing --cov-branch --cov-fail-under=80 -v"
asyncio_mode = "auto"
markers = [
    "pit_check: marks tests as Point-in-Time compliance checks",
]

[tool.mypy]
strict = true
warn_return_any = true
warn_unused_configs = true
packages = ["src"]

[[tool.mypy.overrides]]
module = [
    "MetaTrader5.*",
    "hmmlearn.*",
    "arch.*",
    "vectorbt.*",
    "arcticdb.*",
    "nats.*",
    "xgboost.*",
    "cvxpy.*",
    "msgpack.*",
    "zmq.*",
    "statsmodels.*",
]
ignore_missing_imports = true

[tool.ruff]
target-version = "py312"
line-length = 88
select = ["E","W","F","I","N","UP","B","SIM","ANN","S","C4","DTZ","RUF"]

[tool.ruff.format]
quote-style = "double"

[tool.ruff.lint.per-file-ignores]
"tests/**/*.py" = ["S101", "ANN"]

[tool.coverage.run]
branch = true
source = ["src"]

[tool.coverage.report]
fail_under = 80
show_missing = true
exclude_lines = [
    "pragma: no cover",
    "if TYPE_CHECKING:",
    "@abstractmethod",
    "pass",
]
```

2. Create `Makefile` with targets:
```makefile
.PHONY: lint typecheck test test-integration validate all

lint:
	.venv/bin/ruff check . && .venv/bin/ruff format --check .

typecheck:
	.venv/bin/mypy src/ --strict

test:
	.venv/bin/pytest tests/ --cov=src --cov-fail-under=80 --cov-branch -v

test-integration:
	.venv/bin/pytest tests/integration/ -v --timeout=60

validate:
	.venv/bin/python scripts/ast_validator.py --stubs stubs/ --source src/
	.venv/bin/python scripts/pit_validator.py --source src/alpha/

all: lint typecheck validate test
```

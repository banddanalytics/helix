```python
from __future__ import annotations

import asyncio
from typing import Any
from collections.abc import Callable

import numpy as np

from src.execution.abstract import (
    Bar, MarketDataProvider, OrderExecutor, OrderRequest, OrderResult,
    OrderType, Position, PositionManager, Side, Tick,
)

# MT5 imported conditionally to allow Linux CI to run
try:
    import MetaTrader5 as mt5  # type: ignore[import-untyped]
except ImportError:
    mt5 = None  # type: ignore[assignment]


TIMEFRAME_MAP: dict[str, Any] = {
    "1m": "TIMEFRAME_M1",
    "5m": "TIMEFRAME_M5",
    "15m": "TIMEFRAME_M15",
    "30m": "TIMEFRAME_M30",
    "1h": "TIMEFRAME_H1",
    "4h": "TIMEFRAME_H4",
    "1d": "TIMEFRAME_D1",
    "1w": "TIMEFRAME_W1",
}


class MT5Adapter(MarketDataProvider, OrderExecutor, PositionManager):
    def __init__(
        self,
        account: int,
        password: str,
        server: str,
        mt5_path: str | None = None,
    ) -> None: ...

    async def connect(self) -> None: ...
    async def disconnect(self) -> None: ...
```

Key implementation details:
- `connect()`: Calls `await asyncio.to_thread(mt5.initialize, self._mt5_path)` then `await asyncio.to_thread(mt5.login, self._account, password=self._password, server=self._server)`. On failure, calls `mt5.last_error()` and raises `ConnectionError` with the error tuple.
- `get_ticks()`: `await asyncio.to_thread(mt5.copy_ticks_range, symbol, start_ts, end_ts, mt5.COPY_TICKS_ALL)`. Converts structured numpy array rows to `Tick` dataclasses.
- `get_bars()`: `await asyncio.to_thread(mt5.copy_rates_from_pos, symbol, tf_const, 0, count)`. Populates `spread` field from `row['spread'] * symbol_info.point`.
- `submit_order()`: Builds request dict: `{"action": mt5.TRADE_ACTION_DEAL, "symbol": order.symbol, "volume": order.quantity, "type": mt5.ORDER_TYPE_BUY if order.side == Side.BUY else mt5.ORDER_TYPE_SELL, "price": price, "deviation": 20, "magic": 100001, "type_filling": mt5.ORDER_FILLING_IOC, "comment": order.comment}`. Checks `result.retcode != mt5.TRADE_RETCODE_DONE`.
- `close_position()`: Creates reverse market order for the position.
- `subscribe_ticks()`: 10ms polling loop via `asyncio.sleep(0.01)` calling `mt5.symbol_info_tick()`.

All tests mock the `mt5` module entirely using `unittest.mock.MagicMock()` so they run on Linux without MetaTrader5 installed.

```python
from __future__ import annotations

import random
import uuid
from collections.abc import Callable

import numpy as np

from src.execution.abstract import (
    Bar, MarketDataProvider, OrderExecutor, OrderRequest, OrderResult,
    OrderType, Position, PositionManager, Side, Tick,
)


class SimAdapter(MarketDataProvider, OrderExecutor, PositionManager):
    def __init__(
        self,
        initial_equity: float = 100_000.0,
        spread_pips: float = 1.5,
        seed: int = 42,
    ) -> None:
        self._initial_equity = initial_equity
        self._spread_pips = spread_pips
        self._rng = random.Random(seed)  # D-26: fixed seed
        self._positions: dict[str, Position] = {}
        self._realized_pnl: float = 0.0
        self._margin_used: float = 0.0
        self._current_prices: dict[str, float] = {}  # symbol -> mid price
        self._order_counter: int = 0
```

Key implementation details per decisions:
- D-22: Instant fill at requested price (mid price +/- half spread depending on side)
- D-23: Spread cost applied: buy fills at ask (mid + half_spread), sell fills at bid (mid - half_spread)
- D-24: No slippage in Phase 1
- D-25: Stateful -- maintains `_positions` dict, `_realized_pnl`, `_margin_used`
- D-26: `random.Random(seed)` for deterministic behavior
- D-27: Rejection logic: if margin insufficient (`equity * 0.02 < order_margin`), return `OrderResult(success=False, error_message="Insufficient margin")`; if quantity <= 0, return failure

SimAdapter also needs a `set_price(symbol: str, mid_price: float)` method so tests can inject prices.

```python
from __future__ import annotations

from collections import deque

import numpy as np


class SpreadModel:
    def __init__(self, max_history: int = 10_000) -> None:
        self._history: deque[float] = deque(maxlen=max_history)

    def update(self, spread: float) -> None:
        self._history.append(spread)

    @property
    def median(self) -> float:
        if not self._history:
            return 0.0
        return float(np.median(list(self._history)))

    @property
    def p95(self) -> float:
        if not self._history:
            return 0.0
        return float(np.percentile(list(self._history), 95))

    @property
    def volatility(self) -> float:
        if not self._history:
            return 0.0
        return float(np.std(list(self._history)))

    def cost_adjusted_signal(
        self,
        raw_signal: float,
        expected_holding_bars: int,
        avg_bar_range: float,
    ) -> float:
        expected_move = abs(raw_signal) * avg_bar_range * expected_holding_bars
        round_trip_cost = 2.0 * self.median
        if expected_move == 0.0:
            return 0.0
        cost_ratio = round_trip_cost / expected_move
        if cost_ratio > 0.5:
            return 0.0  # Suppressed: spread > 50% of expected profit
        return raw_signal * (1.0 - cost_ratio)
```

```python
from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True)
class CarryResult:
    carry_long: float    # annualized % for long position
    carry_short: float   # annualized % for short position
    net_carry: float     # carry_long + carry_short (one will be negative)


class SwapRateCalculator:
    @staticmethod
    def compute_annualized_carry(
        swap_long: float,
        swap_short: float,
        point: float,
        mid_price: float,
    ) -> CarryResult:
        if mid_price == 0.0:
            return CarryResult(carry_long=0.0, carry_short=0.0, net_carry=0.0)
        carry_long = (swap_long * point * 365) / mid_price * 100
        carry_short = (swap_short * point * 365) / mid_price * 100
        return CarryResult(
            carry_long=carry_long,
            carry_short=carry_short,
            net_carry=carry_long + carry_short,
        )
```

```python
from __future__ import annotations

import math


class LotSizer:
    @staticmethod
    def kelly_to_lots(
        equity: float,
        kelly_fraction: float,
        stop_loss_pips: float,
        pip_value: float,
        volume_min: float = 0.01,
        volume_max: float = 100.0,
        volume_step: float = 0.01,
    ) -> float:
        if kelly_fraction <= 0 or stop_loss_pips <= 0 or pip_value <= 0:
            return 0.0
        risk_amount = equity * kelly_fraction
        raw_lots = risk_amount / (stop_loss_pips * pip_value)
        # Round DOWN to volume_step
        stepped = math.floor(raw_lots / volume_step) * volume_step
        # Clamp
        stepped = max(volume_min, min(volume_max, stepped))
        return round(stepped, 8)  # avoid floating point noise

    @staticmethod
    def compute_pip_value(
        contract_size: float,
        pip_size: float,
        exchange_rate: float = 1.0,
    ) -> float:
        """Pip value in account currency.

        For USD account trading EURUSD: contract_size=100000, pip_size=0.0001, exchange_rate=1.0
        For USD account trading GBPJPY: need conversion via USDJPY rate.
        """
        if exchange_rate == 0.0:
            return 0.0
        return (contract_size * pip_size) / exchange_rate
```

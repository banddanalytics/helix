    1. Create `dashboard/components/providers/trading-stage-provider.tsx` per D-07:
       ```typescript
       "use client";
       import { createContext, useContext, useState, type ReactNode } from "react";

       type TradingStage = "forex" | "futures";

       interface TradingStageContextValue {
         stage: TradingStage;
         setStage: (stage: TradingStage) => void;
         labels: {
           positionUnit: string;
           priceUnit: string;
           carrySource: string;
         };
       }

       const STAGE_LABELS: Record<TradingStage, TradingStageContextValue["labels"]> = {
         forex: { positionUnit: "lots", priceUnit: "pips", carrySource: "swap rate" },
         futures: { positionUnit: "contracts", priceUnit: "ticks", carrySource: "term structure carry" },
       };

       const TradingStageContext = createContext<TradingStageContextValue | null>(null);

       export function TradingStageProvider({ children }: { children: ReactNode }) {
         const [stage, setStage] = useState<TradingStage>("forex");
         return (
           <TradingStageContext.Provider value={{ stage, setStage, labels: STAGE_LABELS[stage] }}>
             {children}
           </TradingStageContext.Provider>
         );
       }

       export function useTradingStage() {
         const ctx = useContext(TradingStageContext);
         if (!ctx) throw new Error("useTradingStage must be used within TradingStageProvider");
         return ctx;
       }
       ```

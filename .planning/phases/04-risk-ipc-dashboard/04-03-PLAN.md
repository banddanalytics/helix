    2. Create `src/risk/kelly/criterion.py`:
       ```python
       def compute_kelly_fraction(
           returns: np.ndarray,
           max_fraction: float = 0.15,
       ) -> float:
           mu = np.mean(returns)
           var = np.var(returns)
           if var == 0 or mu <= 0:
               return 0.0
           full_kelly = mu / var
           return min(full_kelly, max_fraction)
       ```

    3. Create `src/risk/kelly/regime_adjusted.py`:

       ```python
       REGIME_MULTIPLIERS: dict[RegimeState, float] = {
           RegimeState.TRENDING: 0.5,
           RegimeState.MEAN_REVERTING: 0.4,
           RegimeState.CRISIS: 0.1,
       }

       def apply_regime_kelly(
           returns: np.ndarray,
           regime: RegimeState,
           max_fraction: float = 0.15,
           kelly_scale: float = 1.0,
       ) -> float:
           from src.risk.kelly.criterion import compute_kelly_fraction
           base = compute_kelly_fraction(returns, max_fraction=1.0)  # uncapped
           multiplier = REGIME_MULTIPLIERS.get(regime, 0.3)
           adjusted = base * multiplier * kelly_scale
           return min(adjusted, max_fraction)
       ```

       ```python
       import asyncio
       import json
       import logging
       from typing import Callable
       import nats

       logger = logging.getLogger("helix.ipc")

       class TelemetryPublisher:
           """Publishes TelemetryState snapshots to NATS JetStream on asyncio timers (D-12, IPC-02).

           Uses monotonic clock tracking to avoid timer drift (Pitfall 4 from RESEARCH.md):
           record t0 before processing, sleep for max(0, interval - elapsed).
           """
           def __init__(self, state: TelemetryState, nats_url: str = "nats://localhost:4222") -> None:
               self._state = state
               self._nats_url = nats_url
               self._nc: nats.NATS | None = None
               self._js = None
               self._tasks: list[asyncio.Task] = []

           async def start(self) -> None:
               self._nc = await nats.connect(self._nats_url)
               self._js = self._nc.jetstream()
               # Create TELEMETRY stream if not exists
               try:
                   await self._js.add_stream(
                       name="TELEMETRY",
                       subjects=["telemetry.>"],
                       retention="limits",
                       max_age=7 * 24 * 60 * 60 * 10**9,  # 7 days in nanoseconds
                   )
               except Exception:
                   pass  # Stream may already exist
               # Start publish loops per IPC-02 intervals
               self._tasks.append(asyncio.create_task(self._publish_loop("telemetry.pnl", 0.1, self._state.to_pnl_payload)))
               self._tasks.append(asyncio.create_task(self._publish_loop("telemetry.positions", 1.0, self._state.to_positions_payload)))
               self._tasks.append(asyncio.create_task(self._publish_loop("telemetry.risk", 1.0, self._state.to_risk_payload)))
               self._tasks.append(asyncio.create_task(self._publish_loop("telemetry.regime", 5.0, self._state.to_regime_payload)))

           async def stop(self) -> None:
               for t in self._tasks:
                   t.cancel()
               if self._nc:
                   await self._nc.close()

           async def publish_order_event(self, order: dict) -> None:
               """Publish a single order event immediately (on-event, not timer)."""
               if self._js:
                   await self._js.publish("telemetry.orders", json.dumps(order).encode())

           async def _publish_loop(self, subject: str, interval: float, payload_fn: Callable) -> None:
               loop = asyncio.get_event_loop()
               while True:
                   t0 = loop.time()
                   try:
                       payload = payload_fn()
                       await self._js.publish(subject, json.dumps(payload).encode())
                   except Exception as exc:
                       logger.warning("TelemetryPublisher: failed to publish %s: %s", subject, exc)
                   elapsed = loop.time() - t0
                   await asyncio.sleep(max(0.0, interval - elapsed))
       ```

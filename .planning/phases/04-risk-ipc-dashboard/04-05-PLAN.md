       class EquityCurveTrader:
           def __init__(self, ma_window: int = 50, deriv_window: int = 20, recovery_bars: int = 10) -> None:
               self.ma_window = ma_window
               self.deriv_window = deriv_window
               self.recovery_bars = recovery_bars
               self._sandboxed: dict[str, dict] = {}

           def evaluate(self, strategy_name: str, equity_curve: np.ndarray) -> dict[str, Any]:
               if len(equity_curve) < self.ma_window:
                   return {"action": "HOLD"}

               ma = pd.Series(equity_curve).rolling(self.ma_window).mean().values
               deriv = np.diff(equity_curve, prepend=equity_curve[0])
               deriv_ma = pd.Series(deriv).rolling(self.deriv_window).mean().values

               current_eq = equity_curve[-1]
               is_sandboxed = strategy_name in self._sandboxed

               if not is_sandboxed:
                   if current_eq < ma[-1] and deriv[-1] < deriv_ma[-1]:
                       self._sandboxed[strategy_name] = {"recovery_count": 0}
                       return {"action": "SANDBOX"}
               else:
                   if current_eq > ma[-1] and deriv[-1] > 0:
                       self._sandboxed[strategy_name]["recovery_count"] += 1
                       if self._sandboxed[strategy_name]["recovery_count"] >= self.recovery_bars:
                           del self._sandboxed[strategy_name]
                           return {"action": "RESTORE", "scale": 0.5}
                   else:
                       self._sandboxed[strategy_name]["recovery_count"] = 0

               return {"action": "HOLD"}

           def is_sandboxed(self, strategy_name: str) -> bool:
               return strategy_name in self._sandboxed

           def get_recovery_count(self, strategy_name: str) -> int:
               return self._sandboxed.get(strategy_name, {}).get("recovery_count", 0)
       ```

       class DrawdownMonitor:
           def __init__(self, initial_equity: float = 0.0) -> None:
               self._high_water: float = initial_equity
               self._current_equity: float = initial_equity

           def update(self, equity: float) -> float:
               if equity > self._high_water:
                   self._high_water = equity
               self._current_equity = equity
               if self._high_water <= 0:
                   return 0.0
               return (self._high_water - equity) / self._high_water

           def reset(self, current_equity: float) -> None:
               """Daily reset at midnight UTC (D-20)."""
               self._high_water = current_equity
               self._current_equity = current_equity

           @property
           def drawdown_pct(self) -> float:
               if self._high_water <= 0:
                   return 0.0
               return (self._high_water - self._current_equity) / self._high_water
       ```

       ```python
       import enum
       from datetime import datetime, timedelta, timezone
       import logging

       logger = logging.getLogger("helix.risk")

       class CircuitBreakerLevel(enum.IntEnum):
           L0 = 0  # Normal
           L1 = 1  # Warning: 2% DD, 50% Kelly
           L2 = 2  # Halt: 5% DD, flatten + 1h pause
           L3 = 3  # Shutdown: 10% DD, manual restart

       class CircuitBreaker:
           L1_THRESHOLD = 0.02
           L2_THRESHOLD = 0.05
           L3_THRESHOLD = 0.10
           L2_PAUSE_DURATION = timedelta(hours=1)

           def __init__(self) -> None:
               self._level = CircuitBreakerLevel.L0
               self._l2_pause_until: datetime | None = None
               self._l3_triggered = False

           @property
           def current_level(self) -> CircuitBreakerLevel:
               return self._level

           def evaluate(self, drawdown_pct: float) -> CircuitBreakerLevel:
               """Evaluate drawdown and potentially escalate circuit breaker level.

               Returns the new level. Callers check return value to decide actions:
               - L1: scale Kelly to 50%
               - L2: flatten all positions, pause 1 hour
               - L3: flatten all, disable strategies, require manual restart
               """
               if drawdown_pct >= self.L3_THRESHOLD:
                   if not self._l3_triggered:  # Idempotent (D-19)
                       self._l3_triggered = True
                       self._level = CircuitBreakerLevel.L3
                       logger.critical("Circuit breaker L3: %.1f%% drawdown — SHUTDOWN", drawdown_pct * 100)
                   return self._level

               if drawdown_pct >= self.L2_THRESHOLD and self._level < CircuitBreakerLevel.L2:
                   self._level = CircuitBreakerLevel.L2
                   self._l2_pause_until = datetime.now(timezone.utc) + self.L2_PAUSE_DURATION
                   logger.error("Circuit breaker L2: %.1f%% drawdown — HALT for 1 hour", drawdown_pct * 100)
                   return self._level

               if drawdown_pct >= self.L1_THRESHOLD and self._level < CircuitBreakerLevel.L1:
                   self._level = CircuitBreakerLevel.L1
                   logger.warning("Circuit breaker L1: %.1f%% drawdown — 50%% Kelly", drawdown_pct * 100)
                   return self._level

               return self._level

           def is_paused(self) -> bool:
               if self._level == CircuitBreakerLevel.L2 and self._l2_pause_until:
                   return datetime.now(timezone.utc) < self._l2_pause_until
               return False

           def manual_restart(self) -> bool:
               if self._level == CircuitBreakerLevel.L3:
                   self._level = CircuitBreakerLevel.L0
                   self._l3_triggered = False
                   logger.info("Circuit breaker L3: manual restart — returning to L0")
                   return True
               return False

           def daily_reset(self) -> None:
               """Reset L1 at midnight UTC (D-20). Does NOT clear L2 (if paused) or L3."""
               if self._level == CircuitBreakerLevel.L1:
                   self._level = CircuitBreakerLevel.L0
                   logger.info("Circuit breaker L1: daily reset — returning to L0")
       ```

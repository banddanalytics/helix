       ```typescript
       import { connect, type NatsConnection, StringCodec } from "nats.ws";

       const sc = StringCodec();
       const MAX_RECONNECT_ATTEMPTS = 5;
       const INITIAL_RECONNECT_DELAY_MS = 1000;

       let nc: NatsConnection | null = null;
       let reconnectAttempt = 0;

       export async function getNatsConnection(): Promise<NatsConnection> {
         if (nc && !nc.isClosed()) return nc;

         const url = process.env.NEXT_PUBLIC_NATS_WS_URL || "ws://localhost:9222";

         while (reconnectAttempt < MAX_RECONNECT_ATTEMPTS) {
           try {
             nc = await connect({ servers: url });
             reconnectAttempt = 0; // reset on success

             // Listen for disconnect to trigger reconnect
             (async () => {
               for await (const status of nc!.status()) {
                 if (status.type === "disconnect" || status.type === "error") {
                   nc = null;
                   break;
                 }
               }
             })();

             return nc;
           } catch (err) {
             reconnectAttempt++;
             if (reconnectAttempt >= MAX_RECONNECT_ATTEMPTS) {
               throw new Error(
                 `NATS connection failed after ${MAX_RECONNECT_ATTEMPTS} attempts: ${err}`
               );
             }
             // Exponential backoff: 1s, 2s, 4s, 8s, 16s
             const delay = INITIAL_RECONNECT_DELAY_MS * Math.pow(2, reconnectAttempt - 1);
             await new Promise((r) => setTimeout(r, delay));
           }
         }

         throw new Error("NATS connection exhausted retries");
       }

       export function resetReconnectCounter(): void {
         reconnectAttempt = 0;
       }

       export { sc, type NatsConnection };
       ```

    3. Create `dashboard/components/hooks/use-nats-subscription.ts` per IPC-06 with reconnect awareness:
       ```typescript
       "use client";
       import { useState, useEffect, useRef, useCallback } from "react";
       import { getNatsConnection, sc, resetReconnectCounter } from "@/lib/nats-client";
       import type { NatsConnection, Subscription } from "nats.ws";

       type ConnectionState = "connecting" | "connected" | "stale" | "error";

       interface UseNatsResult<T> {
         data: T | null;
         connectionState: ConnectionState;
         lastUpdate: number | null;
       }

       export function useNatsSubscription<T = unknown>(
         subject: string,
         staleThresholdMs: number = 0, // 0 = auto (2x expected interval)
       ): UseNatsResult<T> {
         const [data, setData] = useState<T | null>(null);
         const [connectionState, setConnectionState] = useState<ConnectionState>("connecting");
         const [lastUpdate, setLastUpdate] = useState<number | null>(null);
         const bufferRef = useRef<T[]>([]);

         useEffect(() => {
           let nc: NatsConnection;
           let sub: Subscription;
           let flushInterval: ReturnType<typeof setInterval>;
           let staleInterval: ReturnType<typeof setInterval>;
           let cancelled = false;

           // 100ms flush interval per IPC-06 — latest message wins
           flushInterval = setInterval(() => {
             if (bufferRef.current.length > 0) {
               const latest = bufferRef.current[bufferRef.current.length - 1];
               setData(latest);
               setLastUpdate(Date.now());
               setConnectionState("connected");
               bufferRef.current = [];
             }
           }, 100);

           // Stale detection: check every second
           const threshold = staleThresholdMs || getDefaultStaleThreshold(subject);
           staleInterval = setInterval(() => {
             // ... check if lastUpdate is older than threshold
           }, 1000);

           const connectAndSubscribe = async () => {
             try {
               nc = await getNatsConnection();
               if (cancelled) return;
               setConnectionState("connecting");
               sub = nc.subscribe(subject);  // Core NATS subscribe (not JetStream pull) per Pitfall 3
               for await (const msg of sub) {
                 if (cancelled) break;
                 try {
                   bufferRef.current.push(JSON.parse(sc.decode(msg.data)) as T);
                 } catch {}
               }
               // If subscription ends (server disconnect), attempt reconnect
               if (!cancelled) {
                 setConnectionState("error");
                 // Auto-reconnect: reset counter and retry after brief delay (IPC-03)
                 resetReconnectCounter();
                 setTimeout(() => {
                   if (!cancelled) connectAndSubscribe();
                 }, 1000);
               }
             } catch (err) {
               if (!cancelled) setConnectionState("error");
             }
           };

           connectAndSubscribe();

           return () => {
             cancelled = true;
             clearInterval(flushInterval);
             clearInterval(staleInterval);
             sub?.unsubscribe();
           };
         }, [subject, staleThresholdMs]);

         return { data, connectionState, lastUpdate };
       }

       function getDefaultStaleThreshold(subject: string): number {
         // 2x expected interval per UI-SPEC Data States
         if (subject.includes("pnl")) return 200;       // 100ms * 2
         if (subject.includes("positions")) return 2000; // 1s * 2
         if (subject.includes("risk")) return 2000;      // 1s * 2
         if (subject.includes("regime")) return 10000;   // 5s * 2
         return 5000; // default
       }
       ```

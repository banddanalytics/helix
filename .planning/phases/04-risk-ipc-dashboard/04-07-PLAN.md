       ```python
       from __future__ import annotations
       import logging
       from typing import Callable
       import numpy as np

       from src.alpha.signal_types import RegimeState, SignalRow
       from src.ipc.telemetry_state import TelemetryState
       from src.risk.circuit_breakers.drawdown_monitor import DrawdownMonitor
       from src.risk.circuit_breakers.kill_switch import CircuitBreaker, CircuitBreakerLevel
       from src.risk.cvar.historical import cvar_historical
       from src.risk.cvar.parametric import cvar_parametric
       from src.risk.cvar.cornish_fisher import cvar_cornish_fisher
       from src.risk.cvar.portfolio_opt import optimize_portfolio_cvar
       from src.risk.ect.equity_curve import EquityCurveTrader
       from src.risk.kelly.regime_adjusted import apply_regime_kelly

       logger = logging.getLogger("helix.risk")

       class RiskEngine:
           """Central risk engine — single entry point for all risk decisions (D-09).

           Gate chain order per D-10:
           1. Circuit breaker state -> reject if L3, scale if L1
           2. ECT sandbox check -> route sandboxed to virtual executor
           3. CVaR budget gate -> optimize_portfolio_cvar weights (D-10 Gate 3)
           4. Kelly fraction -> scale position size, apply ECT restoration scale
           """

           def __init__(
               self,
               telemetry: TelemetryState,
               strategy_returns_fn: Callable[[str], np.ndarray] | None = None,
           ) -> None:
               self._telemetry = telemetry
               self._strategy_returns_fn = strategy_returns_fn
               self._circuit_breaker = CircuitBreaker()
               self._drawdown_monitor = DrawdownMonitor()
               self._ect = EquityCurveTrader()
               self._strategy_equity: dict[str, list[float]] = {}

           def size_signals(
               self,
               signals: list[SignalRow],
               regime: RegimeState,
               equity: float,
           ) -> list[SignalRow]:
               """Apply gate chain and return sized signals (D-10)."""
               # Gate 1: Circuit breaker
               cb_level = self._circuit_breaker.current_level
               if cb_level == CircuitBreakerLevel.L3:
                   logger.info("RiskEngine: L3 active — all signals rejected")
                   return []
               if self._circuit_breaker.is_paused():
                   logger.info("RiskEngine: L2 pause active — all signals rejected")
                   return []

               # Gate 2: ECT sandbox routing
               live_signals = []
               for s in signals:
                   if self._ect.is_sandboxed(s.engine):
                       logger.debug("RiskEngine: %s sandboxed — routing to virtual", s.engine)
                       continue  # sandboxed signals go to virtual executor (handled externally)
                   live_signals.append(s)

               if not live_signals:
                   return []

               # Gate 3: CVaR budget gate (D-10)
               # Collect returns matrix for all live signal engines
               engine_names = list(dict.fromkeys(s.engine for s in live_signals))  # unique, order-preserved
               returns_columns = []
               for engine in engine_names:
                   ret = self._get_strategy_returns(engine)
                   if ret is not None and len(ret) > 0:
                       returns_columns.append(ret)
                   else:
                       # No returns available — use zeros (conservative, gets near-zero weight)
                       returns_columns.append(np.zeros(100))

               # Build returns matrix (rows=time, cols=strategies) and optimize
               min_len = min(len(r) for r in returns_columns)
               if min_len > 1 and len(returns_columns) > 0:
                   returns_matrix = np.column_stack([r[-min_len:] for r in returns_columns])
                   try:
                       cvar_weights = optimize_portfolio_cvar(
                           returns_matrix, alpha=0.95, max_weight=0.25, cvar_budget=0.05
                       )
                   except Exception as exc:
                       logger.warning("RiskEngine: CVaR optimization failed (%s), using equal weights", exc)
                       cvar_weights = np.ones(len(engine_names)) / len(engine_names)
               else:
                   cvar_weights = np.ones(len(engine_names)) / len(engine_names)

               # Map engine -> CVaR weight
               engine_weight = dict(zip(engine_names, cvar_weights))

               # Gate 4: Kelly sizing with ECT restoration scale
               kelly_scale = 0.5 if cb_level == CircuitBreakerLevel.L1 else 1.0
               sized = []
               for s in live_signals:
                   returns = self._get_strategy_returns(s.engine)
                   if returns is not None and len(returns) > 0:
                       fraction = apply_regime_kelly(returns, regime, kelly_scale=kelly_scale)
                   else:
                       fraction = 0.0

                   # Apply CVaR weight from Gate 3
                   weight = engine_weight.get(s.engine, 0.0)

                   # Apply ECT progressive restoration scale (RISK-05)
                   restoration_scale = self._ect.get_restoration_scale(s.engine)
                   self._ect.tick_restoration(s.engine)

                   # Final strength = Kelly fraction * CVaR weight * restoration scale
                   final_strength = fraction * weight * restoration_scale

                   sized.append(SignalRow(
                       symbol=s.symbol,
                       engine=s.engine,
                       direction=s.direction,
                       strength=final_strength,
                       regime=s.regime,
                       z_score=s.z_score,
                       ml_prob=s.ml_prob,
                       carry_rank=s.carry_rank,
                   ))
                   logger.debug(
                       "RiskEngine: %s/%s sized at %.4f (kelly=%.4f, cvar_w=%.4f, restore=%.2f)",
                       s.engine, s.symbol, final_strength, fraction, weight, restoration_scale,
                   )

               # Update telemetry
               self._telemetry.circuit_breaker_level = cb_level.value
               self._telemetry.sandboxed_strategies = [
                   name for name in self._ect._sandboxed
               ]

               return sized

           def on_bar_end(self, equity: float, returns: np.ndarray | None = None) -> CircuitBreakerLevel:
               """Update drawdown, evaluate circuit breaker, and compute CVaR for telemetry.

               Args:
                   equity: Current portfolio equity.
                   returns: Recent portfolio returns array for CVaR computation.
                       If None, CVaR telemetry is not updated.
               """
               dd = self._drawdown_monitor.update(equity)
               level = self._circuit_breaker.evaluate(dd)

               # Update telemetry — equity, drawdown, circuit breaker
               self._telemetry.equity = equity
               self._telemetry.drawdown_pct = dd
               self._telemetry.circuit_breaker_level = level.value

               # Compute CVaR via all 3 methods and populate telemetry (key_link)
               if returns is not None and len(returns) > 10:
                   try:
                       self._telemetry.cvar = {
                           "historical": float(cvar_historical(returns, alpha=0.95)),
                           "parametric": float(cvar_parametric(returns, alpha=0.95)),
                           "cf": float(cvar_cornish_fisher(returns, alpha=0.95)),
                       }
                   except Exception as exc:
                       logger.warning("RiskEngine: CVaR computation failed: %s", exc)
               # If returns is None, leave cvar unchanged (may be stale but not empty)

               return level

           def evaluate_ect(self, strategy_name: str, equity_value: float) -> dict:
               """Evaluate ECT for a strategy. Call per strategy per bar."""
               if strategy_name not in self._strategy_equity:
                   self._strategy_equity[strategy_name] = []
               self._strategy_equity[strategy_name].append(equity_value)
               curve = np.array(self._strategy_equity[strategy_name])
               return self._ect.evaluate(strategy_name, curve)

           def manual_restart(self) -> bool:
               """Manual restart from L3 circuit breaker."""
               return self._circuit_breaker.manual_restart()

           def daily_reset(self, current_equity: float) -> None:
               """Daily midnight UTC reset."""
               self._drawdown_monitor.reset(current_equity)
               self._circuit_breaker.daily_reset()

           def _get_strategy_returns(self, engine: str) -> np.ndarray | None:
               if self._strategy_returns_fn:
                   try:
                       return self._strategy_returns_fn(engine)
                   except Exception:
                       return None
               return None
       ```
